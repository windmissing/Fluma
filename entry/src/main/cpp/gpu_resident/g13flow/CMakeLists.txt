cmake_minimum_required(VERSION 3.16)
project(grs_g13 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(GRS_G13_TEST_HOOKS "Separate diagnostic fault-injection identity" OFF)
include("${CMAKE_CURRENT_LIST_DIR}/kernels.cmake")
if(NOT TARGET grs_cold_slots)
 add_library(grs_cold_slots SHARED "${CMAKE_CURRENT_LIST_DIR}/../cold_slot_broker.cpp")
endif()
add_library(grs_g13_core STATIC kernel_assets.cpp transport.cpp controller.cpp cold.cpp scene.cpp frame_targets.cpp)
set_target_properties(grs_g13_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(grs_g13_core PRIVATE -fno-fast-math -ffp-contract=off)
target_include_directories(grs_g13_core BEFORE PUBLIC "${G13_ASSET_DIR}" "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(grs_g13_core PUBLIC grs_cold_slots libEGL.so libGLESv3.so libhilog_ndk.z.so)
add_library(grs_g13 SHARED bridge.cpp)
target_link_libraries(grs_g13 PUBLIC grs_g13_core libace_napi.z.so libace_ndk.z.so libnative_window.so)
