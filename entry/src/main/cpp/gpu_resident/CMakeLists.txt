# T236 foundation slice; no M0 route is enabled by compiling this library.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/integer_primitives.glsl")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/integer_primitives.glsl" GRS_PRIMITIVES)
string(SHA256 GRS_PRIMITIVE_SHA "${GRS_PRIMITIVES}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/grs_primitive_source.h" "#pragma once\nstatic constexpr const char* kGrsPrimitiveSource=R\"GRS(${GRS_PRIMITIVES})GRS\";\nstatic constexpr const char* kGrsPrimitiveSha=\"${GRS_PRIMITIVE_SHA}\";\n")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/m0_source.comp" "${CMAKE_CURRENT_SOURCE_DIR}/m0_quantum.glsl" "${CMAKE_CURRENT_SOURCE_DIR}/m0_source_root.glsl" "${CMAKE_CURRENT_SOURCE_DIR}/layout_generated.glsl")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/m0_source.comp" GRS_SOURCE)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/m0_source_root.glsl" GRS_SOURCE_ROOT)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/m0_quantum.glsl" GRS_QUANTUM)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/layout_generated.glsl" GRS_LAYOUT)
foreach(GRS_NAME GRS_SOURCE GRS_SOURCE_ROOT GRS_LAYOUT GRS_PRIMITIVES GRS_QUANTUM)
 string(REPLACE "\r\n" "\n" ${GRS_NAME} "${${GRS_NAME}}")
endforeach()
string(REPLACE "/* GRS_LAYOUT */" "${GRS_LAYOUT}" GRS_SOURCE "${GRS_SOURCE}")
string(REPLACE "/* GRS_INTEGER */" "${GRS_PRIMITIVES}" GRS_SOURCE "${GRS_SOURCE}")
string(REPLACE "/* GRS_SOURCE_ROOT */" "${GRS_SOURCE_ROOT}" GRS_SOURCE "${GRS_SOURCE}")
string(REPLACE "/* GRS_QUANTUM */" "${GRS_QUANTUM}" GRS_SOURCE "${GRS_SOURCE}")
string(SHA256 GRS_SOURCE_SHA "${GRS_SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/grs_source_kernel.h" "#pragma once\nstatic constexpr const char* kGrsSourceKernel=R\"GRSS(${GRS_SOURCE})GRSS\";\nstatic constexpr const char* kGrsSourceSha=\"${GRS_SOURCE_SHA}\";\n")
set(GRS_SCENE_HEADER "#pragma once\n")
foreach(GRS_PAIR "m0_scene_full.vert|kGrsFullVertex" "m0_scene.frag|kGrsSceneFragment" "m0_scene_particle.vert|kGrsParticleVertex" "m0_scene_particle.frag|kGrsParticleFragment")
 string(REPLACE "|" ";" GRS_PARTS "${GRS_PAIR}")
 list(GET GRS_PARTS 0 GRS_FILE)
 list(GET GRS_PARTS 1 GRS_SYMBOL)
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${GRS_FILE}")
 file(READ "${CMAKE_CURRENT_SOURCE_DIR}/${GRS_FILE}" GRS_TEXT)
 string(REPLACE "\r\n" "\n" GRS_TEXT "${GRS_TEXT}")
 string(APPEND GRS_SCENE_HEADER "static constexpr const char* ${GRS_SYMBOL}=R\"GRSC(${GRS_TEXT})GRSC\";\n")
endforeach()
string(SHA256 GRS_SCENE_SHA "${GRS_SCENE_HEADER}")
string(APPEND GRS_SCENE_HEADER "static constexpr const char* kGrsSceneShaderSha=\"${GRS_SCENE_SHA}\";\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/grs_scene_sources.h" "${GRS_SCENE_HEADER}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/m1/source_core.comp")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/m1/source_core.comp" M1_SOURCE_CORE)
string(REPLACE "\r\n" "\n" M1_SOURCE_CORE "${M1_SOURCE_CORE}")
string(REPLACE "/* INTEGER_PRIMITIVES */" "${GRS_PRIMITIVES}" M1_SOURCE_CORE "${M1_SOURCE_CORE}")
string(SHA256 M1_SOURCE_CORE_SHA "${M1_SOURCE_CORE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/source_core_kernel.h" "#pragma once\nstatic constexpr const char* kM1SourceCore=R\"M1SC(${M1_SOURCE_CORE})M1SC\";\nstatic constexpr const char* kM1SourceCoreSha=\"${M1_SOURCE_CORE_SHA}\";\n")
add_library(grs_foundation STATIC integer.cpp control.cpp resources.cpp foundation.cpp product_wire.cpp source_transport.cpp scene.cpp run.cpp m1/profile.cpp m1/source_kernel.cpp)
target_include_directories(grs_foundation PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(grs_foundation PRIVATE -fno-fast-math -ffp-contract=off)
set_target_properties(grs_foundation PROPERTIES POSITION_INDEPENDENT_CODE ON)
