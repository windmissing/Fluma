cmake_minimum_required(VERSION 3.16)
project(grs_g9rx LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(GRS_G9RX_TEST_HOOKS "Dedicated native harness; never enabled in product" OFF)
get_filename_component(RX_REPO "${CMAKE_CURRENT_LIST_DIR}/../../../../../.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(RX_ASSET_DIR "${CMAKE_CURRENT_BINARY_DIR}/rx_assets")
file(GLOB RX_INPUTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/*.h" "${CMAKE_CURRENT_LIST_DIR}/*.cpp" "${CMAKE_CURRENT_LIST_DIR}/*.glsl" "${CMAKE_CURRENT_LIST_DIR}/frozen/*" "${RX_REPO}/docs/architecture-T641-*.json")
file(GLOB RX_SHARED_INPUTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/../napi_*.h" "${CMAKE_CURRENT_LIST_DIR}/../program_cache/*.h" "${RX_REPO}/entry/src/main/ets/pages/GpuResidentG9Rx.ets" "${RX_REPO}/entry/src/main/ets/physics/*G9Rx*.ets" "${RX_REPO}/entry/src/main/ets/entryability/G9RxResearchAbility.ets" "${RX_REPO}/entry/src/main/ets/workers/G9RxColdWorker.ets")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${RX_SHARED_INPUTS} "${RX_REPO}/docs/design-T637-manifest-v1.json" "${RX_REPO}/docs/architecture-T552-g9-seed-manifest-v2.json" "${RX_REPO}/docs/design-G9-dynamic-mobilization-manifest-v2.json" ${RX_INPUTS} "${RX_REPO}/tools/g9rx/generate_assets.py" "${RX_REPO}/docs/architecture-T647-acceptance-v1.json")
set(RX_HOOK_ARG "")
if(GRS_G9RX_TEST_HOOKS)
 set(RX_HOOK_ARG --test-hooks)
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${RX_REPO}/tools/g9rx/generate_assets.py" --out "${RX_ASSET_DIR}" ${RX_HOOK_ARG} RESULT_VARIABLE RX_RESULT)
if(NOT RX_RESULT EQUAL 0)
 message(FATAL_ERROR "GNR1_GENERATION_FAILED; no stale fallback")
endif()
if(NOT TARGET grs_cold_slots)
 add_library(grs_cold_slots SHARED "${CMAKE_CURRENT_LIST_DIR}/../cold_slot_broker.cpp")
endif()
add_library(grs_g9rx_core STATIC kernel_assets.cpp abi.cpp transport.cpp completion_gate.cpp controller.cpp controller_cold.cpp cold.cpp cold_capture.cpp scene.cpp stage_capability.cpp progress_clock_v3.cpp planning_control.cpp quota_snapshot_v3.cpp ../../p013renderer/fullframe_targets.cpp)
set_target_properties(grs_g9rx_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(grs_g9rx_core PRIVATE -fno-fast-math -ffp-contract=off)
target_include_directories(grs_g9rx_core PUBLIC "${RX_ASSET_DIR}" "${CMAKE_CURRENT_LIST_DIR}" "${CMAKE_CURRENT_LIST_DIR}/..")
target_link_libraries(grs_g9rx_core PUBLIC grs_cold_slots libEGL.so libGLESv3.so libhilog_ndk.z.so)
add_library(grs_g9rx SHARED bridge.cpp)
target_compile_options(grs_g9rx PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(grs_g9rx PUBLIC grs_g9rx_core libace_napi.z.so libace_ndk.z.so libnative_window.so libnative_display_manager.so libdl.so)
include("${CMAKE_CURRENT_LIST_DIR}/../napi_provider_profile.cmake")
grs_bind_napi_provider_sdk(grs_g9rx)
if(GRS_G9RX_TEST_HOOKS)
 target_compile_definitions(grs_g9rx_core PUBLIC GRS_G9RX_TEST_HOOKS=1)
endif()
