cmake_minimum_required(VERSION 3.18)
project(p011_gpu_pair_prototype LANGUAGES CXX)
add_executable(p011_gpu_pair_prototype
  pair_driver_v1.cpp pair_prepare_v1.cpp pair_executor_v1.cpp pair_owner_v1.cpp)
target_compile_features(p011_gpu_pair_prototype PRIVATE cxx_std_17)
target_include_directories(p011_gpu_pair_prototype PRIVATE .)
if(NOT WIN32)
  target_link_libraries(p011_gpu_pair_prototype PRIVATE libEGL.so libGLESv3.so)
  add_library(p011_gpu_pair_napi SHARED
    pair_binding_v1.cpp pair_prepare_v1.cpp pair_executor_v1.cpp pair_owner_v1.cpp)
  target_compile_features(p011_gpu_pair_napi PRIVATE cxx_std_17)
  target_compile_options(p011_gpu_pair_napi PRIVATE -fno-fast-math -ffp-contract=off)
  target_link_libraries(p011_gpu_pair_napi PRIVATE libace_napi.z.so libEGL.so libGLESv3.so)
  set_target_properties(p011_gpu_pair_napi PROPERTIES OUTPUT_NAME p011_gpu_pair_prototype)
endif()
if(MSVC)
  target_compile_options(p011_gpu_pair_prototype PRIVATE /fp:strict)
else()
  target_compile_options(p011_gpu_pair_prototype PRIVATE -fno-fast-math -ffp-contract=off)
endif()
